  type Incident = Record {
    id: String
    severity: String
    title: String
    services: List<String>
    region: String
    start_time: String
    status: String
  }

  type Finding = Record {
    category: String
    evidence: String
    confidence: String
    recommendation: String
  }

  type HealthCheck = Record {
    service: String
    status: String
    latency_p99_ms: Int
    error_rate_pct: Float
    region: String
  }

  type Severity = P0 | P1 | P2 | P3 | P4

  agent incident_team["Incident Team"]
    flow detect_and_respond["Detect & Respond"]
      agent sentinel["Sentinel"]
        task detect["Detect Anomaly"]
          region["region"]
          poll_health["poll_health"]
          health_data["Health Data"]
          detect_anomaly["detect_anomaly"]
          alert["Alert"]
          region --> poll_health --> health_data --> detect_anomaly --> alert
        end
      end

      agent monitor["Monitor"]
        task triage["Triage"]
          triage_alert["triage_alert"]
          triage_result["Triage Result"]
          alert --> triage_alert --> triage_result
          triage_alert --o triage_tpl_ref
        end
      end

      agent investigator["Investigator"]
        task investigate["Investigate"]
          analyze_incident["analyze_incident"]
          investigation["Investigation Report"]
          triage_result --> analyze_incident --> investigation
          analyze_incident --o investigation_tpl_ref
        end
        task rca["Root Cause Analysis"]
          root_cause_analysis["root_cause_analysis"]
          root_cause["Root Cause"]
          investigation --> root_cause_analysis --> root_cause
        end
        investigate --> rca
      end

      agent responder["Responder"]
        task mitigate["Generate Runbook"]
          generate_runbook["generate_runbook"]
          runbook["Runbook"]
          root_cause --> generate_runbook --> runbook
          generate_runbook --o runbook_tpl_ref
        end
      end

      agent reporter["Reporter"]
        task report["Create Dashboard"]
          create_report["create_report"]
          dashboard["Incident Dashboard"]
          runbook --> create_report --> dashboard
          triage_result --> create_report
          investigation --> create_report
          root_cause --> create_report
          create_report --- dashboard_style
        end
      end
    end
  end
